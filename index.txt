    <script src="https://code.jquery.com/jquery-3.7.0.min.js"></script>
    <!-- 引入 PapaParse (解析 CSV) -->
    <script src="https://cdnjs.cloudflare.com/ajax/libs/PapaParse/5.4.1/papaparse.min.js"></script>
    <!-- 引入 DataTables JS -->
    <script src="https://cdn.datatables.net/1.13.6/js/jquery.dataTables.min.js"></script>

    <script>
        // 先嘗試不使用 encodeURI，直接輸入完整檔名
        const csvFileName = "data_20260430110928招財進保.csv";

        Papa.parse(csvFileName, {
            download: true,
            header: true,
            skipEmptyLines: true,
            encoding: "Big5",
            complete: function(results) {
                const data = results.data;
                
                if(data.length > 0) {
                    const columns = Object.keys(data[0]).map(function(key) {
                        return { title: key, data: key };
                    });

                    document.getElementById('loadingText').style.display = 'none';
                    document.getElementById('fundTable').style.display = 'table';

                    $('#fundTable').DataTable({
                        data: data,
                        columns: columns,
                        pageLength: 25,
                        order: [[ 11, 'desc' ]],
                        language: {
                            url: "https://cdn.datatables.net/plug-ins/1.13.6/i18n/zh-HANT.json"
                        }
                    });
                } else {
                    document.getElementById('loadingText').innerText = "CSV 檔案內似乎沒有資料哦！";
                }
            },
            error: function(err) {
                document.getElementById('loadingText').innerText = "讀取失敗！如果您是直接點擊檔案開啟，請改用 GitHub Pages 或本機伺服器瀏覽。";
                console.error("CSV 解析錯誤:", err);
            }
        });
    </script>
